## 3D体积渲染

### 功能描述

使用VTK进行高质量的3D体积渲染，支持GPU加速的光线投射算法。

### 渲染特性

- **GPU光线投射**: 使用vtkGPUVolumeRayCastMapper
- **智能采样**: 自适应采样距离
- **颜色映射**: 医学影像专用颜色传输函数
- **不透明度控制**: 分段不透明度传输函数
- **光照效果**: 支持阴影和漫反射

### 核心代码

**文件**: `src/core/volume_render.py`

```python
class VolumeRenderer:
    """3D体积渲染器"""
    
    def create_volume_actor(self) -> Optional[vtk.vtkVolume]:
        """创建体积渲染演员"""
        # 1. 将numpy数组转换为VTK图像数据
        vtk_image = self._numpy_to_vtk_image(self.volume_data)
        vtk_image.SetSpacing(self.spacing)
        vtk_image.SetOrigin(self.origin)
        
        # 2. 创建GPU体积光线投射映射器
        self.volume_mapper = vtk.vtkGPUVolumeRayCastMapper()
        self.volume_mapper.SetInputData(vtk_image)
        self.volume_mapper.SetBlendModeToComposite()
        
        # 3. 创建体积属性
        self.volume_property = vtk.vtkVolumeProperty()
        self.volume_property.ShadeOn()
        self.volume_property.SetInterpolationTypeToLinear()
        
        # 4. 创建颜色传输函数（医学影像配色）
        color_func = vtk.vtkColorTransferFunction()
        color_func.AddRGBPoint(0.0, 0.0, 0.0, 0.0)      # 黑色 - 背景
        color_func.AddRGBPoint(0.3, 0.5, 0.2, 0.1)      # 棕色 - 软组织
        color_func.AddRGBPoint(0.6, 0.9, 0.6, 0.3)      # 浅棕色 - 肌肉
        color_func.AddRGBPoint(1.0, 1.0, 1.0, 0.9)      # 浅黄色 - 骨骼
        
        # 5. 创建不透明度传输函数
        opacity_func = vtk.vtkPiecewiseFunction()
        opacity_func.AddPoint(0.0, 0.0)   # 完全透明
        opacity_func.AddPoint(0.1, 0.0)   # 背景透明
        opacity_func.AddPoint(0.3, 0.3)   # 软组织半透明
        opacity_func.AddPoint(0.6, 0.6)   # 肌肉较不透明
        opacity_func.AddPoint(1.0, 0.8)   # 骨骼基本不透明
        
        self.volume_property.SetColor(color_func)
        self.volume_property.SetScalarOpacity(opacity_func)
        
        # 6. 创建体积演员
        self.volume = vtk.vtkVolume()
        self.volume.SetMapper(self.volume_mapper)
        self.volume.SetProperty(self.volume_property)
        
        return self.volume
```

## 切割平面功能

### 功能描述

在3D模型中创建切割平面，显示为红色线条轮廓和浅红色填充区域。

### 视觉样式

- **轮廓线**: 红色 (RGB: 0.8, 0.2, 0.2)，线宽3.0，透明度0.7
- **填充区域**: 浅红色 (RGB: 0.9, 0.6, 0.6)，透明度0.7

### 核心代码

**文件**: `src/core/volume_render.py`

```python
def create_cut_plane(self, position_percent: float = 0.5, 
                    normal: Tuple[float, float, float] = (0, 0, 1),
                    return_data: bool = False):
    """创建切割平面（红色线条+浅红色填充）"""
    
    # 1. 将numpy数组转换为VTK图像数据
    vtk_image = self._numpy_to_vtk_image(self.volume_data)
    vtk_image.SetSpacing(self.spacing)
    vtk_image.SetOrigin(self.origin)
    
    # 2. 获取体积边界
    bounds = vtk_image.GetBounds()
    
    # 3. 计算切割位置（根据法线方向）
    if abs(normal[2]) > abs(normal[0]) and abs(normal[2]) > abs(normal[1]):
        # 主要垂直于Z轴
        cut_position = bounds[4] + (bounds[5] - bounds[4]) * position_percent
        plane_origin = [(bounds[0] + bounds[1]) / 2, 
                       (bounds[2] + bounds[3]) / 2, 
                       cut_position]
    # ... 其他方向的处理
    
    # 4. 创建切割平面
    plane = vtk.vtkPlane()
    plane.SetOrigin(plane_origin[0], plane_origin[1], plane_origin[2])
    plane.SetNormal(normal[0], normal[1], normal[2])
    
    # 5. 执行切割操作
    cutter = vtk.vtkCutter()
    cutter.SetCutFunction(plane)
    cutter.SetInputData(vtk_image)
    cutter.Update()
    cut_polydata = cutter.GetOutput()
    
    # 6. 创建红色线条Actor（轮廓线）
    cut_mapper = vtk.vtkPolyDataMapper()
    cut_mapper.SetInputData(cut_polydata)
    
    cut_actor = vtk.vtkActor()
    cut_actor.SetMapper(cut_mapper)
    cut_actor.GetProperty().SetColor(0.8, 0.2, 0.2)  # 红色轮廓
    cut_actor.GetProperty().SetLineWidth(3.0)        # 线宽3.0
    cut_actor.GetProperty().SetOpacity(0.7)          # 透明度0.7
    cut_actor.GetProperty().SetRepresentationToWireframe()  # 线框模式
    
    # 7. 创建浅红色填充平面Actor
    fill_actor = None
    try:
        # 使用vtkFeatureEdges提取轮廓线
        feature_edges = vtk.vtkFeatureEdges()
        feature_edges.SetInputData(cut_polydata)
        feature_edges.BoundaryEdgesOn()
        feature_edges.Update()
        contour_polydata = feature_edges.GetOutput()
        
        if contour_polydata.GetNumberOfLines() > 0:
            # 使用vtkContourTriangulator创建填充平面
            triangulator = vtk.vtkContourTriangulator()
            triangulator.SetInputData(contour_polydata)
            triangulator.Update()
            filled_polydata = triangulator.GetOutput()
            
            if filled_polydata.GetNumberOfPolys() > 0:
                fill_mapper = vtk.vtkPolyDataMapper()
                fill_mapper.SetInputData(filled_polydata)
                
                fill_actor = vtk.vtkActor()
                fill_actor.SetMapper(fill_mapper)
                fill_actor.GetProperty().SetColor(0.9, 0.6, 0.6)  # 浅红色填充
                fill_actor.GetProperty().SetOpacity(0.7)
                fill_actor.GetProperty().SetRepresentationToSurface()
    
    return cut_actor, fill_actor, cut_polydata if return_data else (cut_actor, fill_actor)
```

### 使用示例

```python
from core.volume_render import get_volume_renderer

volume_renderer = get_volume_renderer()

# 创建切割平面（50%位置，垂直于Z轴）
cut_actor, fill_actor = volume_renderer.create_cut_plane(
    position_percent=0.5,
    normal=(0, 0, 1)
)

# 添加到渲染器
renderer.AddActor(cut_actor)
if fill_actor:
    renderer.AddActor(fill_actor)
```

## TEE扇形视图

### 功能描述

模拟经食管超声心动图（TEE）的扇形显示方式，将矩形切片数据转换为扇形显示格式。

### TEE扇形特性

- **扇形角度**: 60°（标准TEE探头角度）
- **探头位置**: 扇形顶点（顶部）
- **深度方向**: 向下展开（远场在下方）
- **近场深度**: 0.0（探头表面）
- **远场深度**: 根据图像尺寸自动计算

### 核心代码

**文件**: `src/gui/cross_section_window.py`

```python
def _create_tee_fan_image(self, source_array, fan_angle, near_depth, far_depth, apex_x, apex_y):
    """创建TEE扇形图像"""
    height, width = source_array.shape
    fan_image = np.zeros_like(source_array)
    
    half_angle_rad = np.radians(fan_angle / 2.0)
    
    # 创建坐标网格
    y_coords, x_coords = np.mgrid[0:height, 0:width]
    
    # 计算相对于扇形顶点的坐标
    dx = x_coords - apex_x
    dy = y_coords - apex_y
    
    # 计算极坐标
    r = np.sqrt(dx**2 + dy**2)  # 距离
    theta = np.arctan2(dx, dy)  # 角度（弧度）
    
    # 只处理扇形区域内的点
    in_fan = (np.abs(theta) <= half_angle_rad) & (r >= near_depth) & (r <= far_depth)
    
    # 将极坐标映射回原始图像坐标
    for y in range(height):
        for x in range(width):
            if in_fan[y, x]:
                # 极坐标 -> 原始图像坐标
                angle_normalized = (theta[y, x] / half_angle_rad + 1) / 2  # 0到1
                depth_normalized = (r[y, x] - near_depth) / (far_depth - near_depth)  # 0到1
                
                src_x = int(angle_normalized * (width - 1))
                src_y = int(depth_normalized * (height - 1))
                
                # 边界检查
                src_x = np.clip(src_x, 0, width - 1)
                src_y = np.clip(src_y, 0, height - 1)
                
                fan_image[y, x] = source_array[src_y, src_x]
    
    return fan_image
```

### 坐标转换

```
矩形图像坐标 -> 极坐标 -> 扇形图像坐标
```

## 渲染模式切换

### 功能描述

支持三种渲染模式：体积渲染、表面渲染、线框渲染。

### 渲染模式对比

| 模式 | 描述 | 适用场景 |
|------|------|----------|
| 体积渲染 | GPU光线投射，显示内部结构 | 观察器官内部 |
| 表面渲染 | Marching Cubes等值面 | 观察器官表面 |
| 线框渲染 | 表面渲染的线框模式 | 观察几何结构 |

### 核心代码

**文件**: `src/gui/vtk_manager.py`

```python
def change_render_mode(self, mode):
    """更改渲染模式"""
    from core.volume_render import get_volume_renderer
    volume_renderer = get_volume_renderer()
    
    # 清除现有演员
    self.renderer.RemoveAllViewProps()
    
    if mode == "体积渲染":
        # 创建体积演员
        volume_actor = volume_renderer.create_volume_actor()
        if volume_actor is not None:
            self.renderer.AddActor(volume_actor)
            self.current_actor = volume_actor
    
    elif mode == "表面渲染":
        # 创建表面演员
        surface_actor = volume_renderer.create_surface_actor(threshold=0.3)
        if surface_actor is not None:
            self.renderer.AddActor(surface_actor)
            self.current_actor = surface_actor
    
    elif mode == "线框渲染":
        # 创建表面演员并设置为线框模式
        surface_actor = volume_renderer.create_surface_actor(threshold=0.3)
        if surface_actor is not None:
            surface_actor.GetProperty().SetRepresentationToWireframe()
            surface_actor.GetProperty().SetColor(0.9, 0.3, 0.2)
            self.renderer.AddActor(surface_actor)
            self.current_actor = surface_actor
    
    # 添加坐标轴
    axes = vtk.vtkAxesActor()
    axes.SetTotalLength(100, 100, 100)
    self.renderer.AddActor(axes)
    
    # 重置相机并重新渲染
    self.renderer.ResetCamera()
    self.vtk_widget.GetRenderWindow().Render()
```

### Marching Cubes算法

**文件**: `src/core/volume_render.py`

```python
def create_surface_actor(self, threshold: float = 0.3) -> Optional[vtk.vtkActor]:
    """创建表面渲染演员"""
    # 1. 将numpy数组转换为VTK图像数据
    vtk_image = self._numpy_to_vtk_image(self.volume_data)
    
    # 2. 创建Marching Cubes算法
    marching_cubes = vtk.vtkMarchingCubes()
    marching_cubes.SetInputData(vtk_image)
    marching_cubes.SetValue(0, threshold)  # 等值面阈值
    marching_cubes.ComputeNormalsOn()
    
    # 3. 创建映射器和演员
    mapper = vtk.vtkPolyDataMapper()
    mapper.SetInputConnection(marching_cubes.GetOutputPort())
    
    actor = vtk.vtkActor()
    actor.SetMapper(mapper)
    actor.GetProperty().SetColor(0.9, 0.3, 0.2)  # 红色
    actor.GetProperty().SetOpacity(0.7)
    actor.GetProperty().SetAmbient(0.1)
    actor.GetProperty().SetDiffuse(0.7)
    actor.GetProperty().SetSpecular(0.5)
    actor.GetProperty().SetSpecularPower(50)
    
    return actor
```

## 透明度控制

### 功能描述

实时调整3D模型的透明度，支持体积渲染和普通演员的透明度控制。

### 透明度范围

- **滑块范围**: 10-100（对应10%-100%）
- **体积渲染**: 使用不透明度传输函数
- **普通演员**: 直接设置Opacity属性

### 核心代码

**文件**: `src/core/volume_render.py`

```python
def update_opacity(self, opacity_value: float):
    """更新体积透明度"""
    if not VTK_AVAILABLE or self.volume_property is None:
        return
    
    # 确保opacity_value在合理范围内
    adjusted_opacity = max(0.1, min(1.0, opacity_value))
    
    # 获取当前的不透明度传输函数
    opacity_func = self.volume_property.GetScalarOpacity()
    
    if opacity_func is None:
        # 创建新的不透明度传输函数
        opacity_func = vtk.vtkPiecewiseFunction()
        opacity_func.AddPoint(0.0, 0.0)
        opacity_func.AddPoint(0.1, 0.0)
        opacity_func.AddPoint(0.3, 0.3 * adjusted_opacity)
        opacity_func.AddPoint(0.6, 0.6 * adjusted_opacity)
        opacity_func.AddPoint(1.0, 1.0 * adjusted_opacity)
        self.volume_property.SetScalarOpacity(opacity_func)
    else:
        # 更新现有点的不透明度
        # 使用温和的缩放方式
        scale_factor = 0.3 + 0.5 * adjusted_opacity  # 0.3到0.8之间
        base_opacity = 0.05 + 0.15 * adjusted_opacity  # 0.05到0.2之间
        
        # 采样关键点并更新
        sample_points = [0.0, 0.1, 0.3, 0.6, 1.0]
        opacity_func.RemoveAllPoints()
        for x in sample_points:
            new_y = x * scale_factor + base_opacity
            new_y = max(0.0, min(1.0, new_y))
            opacity_func.AddPoint(x, new_y)
    
    # 触发重新渲染
    if self.volume is not None:
        self.volume.Modified()
        self.volume.Update()
```

```python
def change_opacity(self, opacity_value):
    """更改模型透明度"""
    opacity = opacity_value / 100.0
    
    from core.volume_render import get_volume_renderer
    volume_renderer = get_volume_renderer()
    
    # 更新体积演员的透明度
    if volume_renderer.volume is not None:
        volume_renderer.update_opacity(opacity)
    
    # 更新所有普通演员的透明度
    actors = self.renderer.GetActors()
    actors.InitTraversal()
    actor = actors.GetNextItem()
    while actor:
        if not isinstance(actor, vtk.vtkVolume):
            actor.GetProperty().SetOpacity(opacity)
        actor = actors.GetNextItem()
    
    self.vtk_widget.GetRenderWindow().Render()
```
